function SmartImage({ source, fallback, size = 70 }) {
  const [hasError, setHasError] = useState(false);
  if (!source.uri) {
    return <Image source={source} style={{ width: size, height: size }} resizeMode="contain" />;
  }
  return (
    <View style={{ width: size, height: size, justifyContent: 'center', alignItems: 'center' }}>
      {!hasError ? (
        <Image source={source} style={{ width: size, height: size }} resizeMode="contain" onError={() => setHasError(true)} />
      ) : (
        <Image source={fallback} style={{ width: size, height: size }} resizeMode="contain" />
      )}
    </View>
  );
}

function ToastNotification({ message, type, visible, onHide }) {
  const translateY = useState(new Animated.Value(100))[0];
  const opacity = useState(new Animated.Value(0))[0];

  useEffect(() => {
    if (visible) {
      Animated.parallel([
        Animated.spring(translateY, { toValue: 0, useNativeDriver: true, friction: 8 }),
        Animated.timing(opacity, { toValue: 1, duration: 200, useNativeDriver: true }),
      ]).start();

      const timer = setTimeout(() => {
        Animated.parallel([
          Animated.timing(translateY, { toValue: 100, duration: 300, useNativeDriver: true }),
          Animated.timing(opacity, { toValue: 0, duration: 300, useNativeDriver: true }),
        ]).start(() => onHide());
      }, 2500);
      return () => clearTimeout(timer);
    }
  }, [visible]);

  if (!visible) return null;

  const getIcon = () => {
    switch (type) {
      case 'success': return { name: 'checkmark-circle', color: '#4CAF50' };
      case 'error': return { name: 'close-circle', color: '#FF4444' };
      case 'info': return { name: 'information-circle', color: '#2196F3' };
      default: return { name: 'information-circle', color: '#FF6B00' };
    }
  };
  const icon = getIcon();

  return (
    <Animated.View style={[styles.toastContainer, { transform: [{ translateY }], opacity }]}>
      <View style={styles.toastContent}>
        <Ionicons name={icon.name} size={28} color={icon.color} />
        <Text style={styles.toastText}>{message}</Text>
      </View>
    </Animated.View>
  );
}

function RequestPartModal({ visible, onClose, onSubmit, defaultCar, defaultPhone }) {

  const [part, setPart] = useState('');
  const [comment, setComment] = useState('');
  const [phone, setPhone] = useState(defaultPhone || '');

  useEffect(() => {
    if (visible) {
      setCar(defaultCar || '');
      setPart('');
      setComment('');
      setPhone(defaultPhone || '');
    }
  }, [visible]);

  const handleSubmit = () => {
    if (part.trim().length < 3) {
      Alert.alert('Ошибка', 'Укажите название или артикул запчасти');
      return;
    }
    if (phone.trim().length < 5) {

      return;
    }

  return (

      <View style={styles.modalOverlay}>
        <View style={styles.modalContent}>

            <Ionicons name="close-circle" size={32} color="#888" />
          </TouchableOpacity>

          <Text style={styles.modalSubtitle}>Не нашли нужную деталь? Напишите нам, и мы постараемся её найти!</Text>
          
          <Text style={styles.inputLabel}>Ваш автомобиль</Text>
          <TextInput style={styles.modalInput} placeholder="Например: Honda Civic 2008" value={car} onChangeText={setCar} />
          
          <Text style={styles.inputLabel}>Название или артикул запчасти *</Text>

function HomeScreen({ onCategoryPress, onViewProduct, searchQuery, onSearchChange, garageCars }) {
  const garageCarNames = getGarageCarNames(garageCars);
  const filteredCategories = useMemo(() => {
    return categories.map(cat => {
      const allProducts = productsByCategory[cat.name] || [];
      const filtered = filterProductsByGarage(allProducts, garageCarNames);
      return { ...cat, filteredCount: filtered.length, hasProducts: filtered.length > 0 };
    }).filter(cat => cat.hasProducts);
  }, [garageCars]);

  const searchResults = useMemo(() => {
    const q = String(searchQuery || '').toLowerCase().trim();
    if (!q) return [];
    const results = [];
    Object.keys(productsByCategory).forEach((catName) => {
      const products = productsByCategory[catName] || [];
      products.forEach((product) => {
        if (product.article.toLowerCase().includes(q) || product.name.toLowerCase().includes(q) || product.car.toLowerCase().includes(q) || (product.keywords && product.keywords.some(k => k.toLowerCase().includes(q)))) {
          results.push({ ...product, categoryName: catName });
        }
      });
    });
    return results;
  }, [searchQuery]);

  return (
    <ScrollView showsVerticalScrollIndicator={false}>
      <View style={styles.header}><Text style={styles.logo}>AUTOPARTS</Text></View>
      {garageCars.length === 0 && (
        <View style={styles.emptyGarageBanner}>
          <Ionicons name="car-outline" size={48} color="#FF6B00" />
          <Text style={styles.emptyGarageTitle}>Добавьте ваш автомобиль</Text>
          <Text style={styles.emptyGarageText}>Перейдите в профиль и добавьте авто</Text>
        </View>
      )}
      {garageCars.length > 0 && (
        <ScrollView horizontal showsHorizontalScrollIndicator={false} style={styles.garageStrip}>
          {garageCars.map((car) => (
            <View key={car.id} style={styles.garageChip}>
              <Ionicons name="car-sport" size={16} color="#FF6B00" />
              <Text style={styles.garageChipText}>{car.name}</Text>
            </View>
          ))}
        </ScrollView>
      )}
      <View style={styles.searchWrapper}>
        <Text style={styles.searchIcon}>🔍</Text>
        <TextInput style={styles.searchInput} placeholder="Поиск по артикулу, названию или авто..." placeholderTextColor="#999" value={searchQuery} onChangeText={onSearchChange} autoCapitalize="none" />
        {searchQuery.length > 0 && <TouchableOpacity onPress={() => onSearchChange('')}><Ionicons name="close-circle" size={20} color="#888" /></TouchableOpacity>}
      </View>

      {searchQuery.trim().length > 0 && (
        <View style={styles.searchResultsWrapper}>
          <Text style={styles.searchResultsTitle}>{searchResults.length > 0 ? `Найдено: ${searchResults.length}` : 'Ничего не найдено'}</Text>
          {searchResults.map((product) => (
            <TouchableOpacity key={product.id} style={styles.searchResultCard} onPress={() => onCategoryPress(product.categoryName, product.id)}>
              <TouchableOpacity onPress={() => onViewProduct(product)} style={styles.searchResultImageWrapper}><SmartImage source={product.image} fallback={product.fallback} size={60} /></TouchableOpacity>
              <View style={styles.searchResultInfo}>
                <Text style={styles.searchResultName}>{product.name}</Text>
                <Text style={styles.searchResultCar}>{product.car}</Text>
                <Text style={styles.searchResultArticle}>Арт: {product.article}</Text>
              </View>
              <Text style={styles.searchResultPrice}>{product.price.toLocaleString('ru-RU')} ₽</Text>
            </TouchableOpacity>
          ))}
        </View>
      )}

      {searchQuery.trim().length === 0 && (
        <>
          {filteredCategories.length > 0 ? (
            <View style={styles.gridWrapper}>
              <View style={styles.grid}>
                {filteredCategories.map((cat) => (
                  <TouchableOpacity key={cat.id} style={styles.card} activeOpacity={0.7} onPress={() => onCategoryPress(cat.name)}>
                    <Image source={cat.icon} style={styles.cardImage} resizeMode="contain" />
                    <Text style={styles.cardName} numberOfLines={1} adjustsFontSizeToFit>{cat.name}</Text>
                    <Text style={styles.cardCount}>{cat.filteredCount}</Text>
                  </TouchableOpacity>
                ))}
              </View>
            </View>
          ) : (
            <View style={styles.emptyState}>
              <Ionicons name="search-outline" size={64} color="#555" />
              <Text style={styles.emptyText}>Нет товаров для ваших авто</Text>
            </View>
          )}
          <View style={{ height: 40 }} />
        </>
      )}
    </ScrollView>
  );
}

function CatalogScreen({ categoryName, onBack, cart, onAddToCart, onRemoveFromCart, highlightProductId, garageCars, onViewProduct, onShowToast }) {

function CheckoutScreen({ cart, onBack, onSubmitOrder, defaultName, defaultPhone, onShowToast }) {
  const [name, setName] = useState(defaultName || '');
  const [phone, setPhone] = useState(defaultPhone || '');
  const [selectedTC, setSelectedTC] = useState('cdek');
  const [address, setAddress] = useState('');
  const [comment, setComment] = useState('');
  const [errors, setErrors] = useState({});
  const total = useMemo(() => cart.reduce((sum, item) => sum + item.price * item.quantity, 0), [cart]);
  const totalItems = useMemo(() => cart.reduce((sum, item) => sum + item.quantity, 0), [cart]);

  const validate = () => {
    const newErrors = {};
    if (name.trim().length < 2) newErrors.name = 'Введите имя';
    const phoneDigits = phone.replace(/\D/g, '');
    if (phoneDigits.length < 10 || phoneDigits.length > 11) newErrors.phone = 'Введите 10-11 цифр';
    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };

  const handleSubmit = () => {
    if (validate()) {
      const tc = transportCompanies.find(t => t.id === selectedTC);
      onSubmitOrder({ name, phone, address, comment, total, totalItems, transportCompany: tc.name, items: [...cart] });
    }
  };

  return (
    <KeyboardAvoidingView style={{ flex: 1 }} behavior={Platform.OS === 'ios' ? 'padding' : undefined}>
      <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={{ paddingBottom: 40 }}>
        <View style={styles.catalogHeader}>
          <TouchableOpacity onPress={onBack} style={styles.backButton}><Ionicons name="arrow-back" size={24} color="#FFF" /><Text style={styles.backText}>Назад</Text></TouchableOpacity>
          <Text style={styles.catalogTitle}>Оформление</Text>
        </View>
        <View style={styles.checkoutWrapper}>
          <View style={styles.checkoutSection}>
            <Text style={styles.checkoutSectionTitle}>Ваш заказ ({totalItems} шт.)</Text>
            {cart.map((item) => (<View key={item.id} style={styles.checkoutItem}><Text style={styles.checkoutItemName}>{item.name}</Text><Text style={styles.checkoutItemDetails}>{item.quantity} шт. × {item.price.toLocaleString('ru-RU')} ₽</Text></View>))}
            <View style={styles.checkoutTotalRow}><Text style={styles.checkoutTotalLabel}>Итого:</Text><Text style={styles.checkoutTotalSum}>{total.toLocaleString('ru-RU')} ₽</Text></View>
          </View>
          <View style={styles.checkoutSection}>
            <Text style={styles.checkoutSectionTitle}>Данные доставки</Text>
            <Text style={styles.inputLabel}>Имя *</Text>
            <TextInput style={[styles.checkoutInput, errors.name && styles.inputError]} placeholder="Иван Иванов" placeholderTextColor="#888" value={name} onChangeText={setName} />
            {errors.name && <Text style={styles.errorText}>{errors.name}</Text>}
            <Text style={styles.inputLabel}>Телефон *</Text>
            <TextInput style={[styles.checkoutInput, errors.phone && styles.inputError]} placeholder="+7 (999) 123-45-67" placeholderTextColor="#888" value={phone} onChangeText={setPhone} keyboardType="phone-pad" />
            {errors.phone && <Text style={styles.errorText}>{errors.phone}</Text>}
            <Text style={styles.inputLabel}>Транспортная компания *</Text>
            <View style={styles.tcList}>
              {transportCompanies.map((tc) => (
                <TouchableOpacity key={tc.id} style={[styles.tcOption, selectedTC === tc.id && styles.tcOptionSelected]} onPress={() => setSelectedTC(tc.id)}>
                  <View style={styles.tcRadio}>{selectedTC === tc.id && <View style={styles.tcRadioDot} />}</View>
                  <View style={styles.tcInfo}><Text style={[styles.tcName, selectedTC === tc.id && styles.tcNameSelected]}>{tc.name}</Text><Text style={styles.tcCity}>{tc.city}</Text></View>
                </TouchableOpacity>
              ))}
            </View>
            <Text style={styles.inputLabel}>Адрес доставки (необязательно)</Text>
            <TextInput style={styles.checkoutInput} placeholder="Улица, дом, квартира" placeholderTextColor="#888" value={address} onChangeText={setAddress} multiline />
            <Text style={styles.inputLabel}>Комментарий (необязательно)</Text>
            <TextInput style={styles.checkoutInput} placeholder="Подъезд, код домофона..." placeholderTextColor="#888" value={comment} onChangeText={setComment} multiline numberOfLines={3} />
          </View>
          <TouchableOpacity style={styles.submitButton} onPress={handleSubmit}><Text style={styles.submitButtonText}>Подтвердить заказ</Text></TouchableOpacity>
        </View>
      </ScrollView>
    </KeyboardAvoidingView>
  );
}

      <Text style={styles.successTitle}>Заказ оформлен!</Text>

    </View>

function CartScreen({ cart, onRemove, onClear, onCheckout, onShowToast }) {
  const total = useMemo(() => cart.reduce((sum, item) => sum + item.price * item.quantity, 0), [cart]);
  const totalItems = useMemo(() => cart.reduce((sum, item) => sum + item.quantity, 0), [cart]);

  if (cart.length === 0) {
    return (<View style={styles.screenContainer}><Ionicons name="cart-outline" size={80} color="#555" /><Text style={styles.screenTitle}>Корзина пуста</Text></View>);
  }

  const renderItem = ({ item }) => (
    <View style={styles.cartItem}>
      <SmartImage source={item.image} fallback={item.fallback} size={60} />
      <View style={styles.cartItemInfo}>
        <Text style={styles.cartItemName}>{item.name}</Text>
        <Text style={styles.cartItemArticle}>Арт: {item.article}</Text>
        <View style={styles.quantityRow}><Text style={styles.quantityLabel}>Кол-во:</Text><Text style={styles.quantityValue}>{item.quantity} шт.</Text></View>
      </View>
      <View style={styles.cartItemAction}>
        <Text style={styles.cartItemPrice}>{(item.price * item.quantity).toLocaleString('ru-RU')} ₽</Text>
        <TouchableOpacity style={styles.removeButton} onPress={() => { onRemove(item.id); onShowToast('Товар удалён из корзины', 'info'); }}><Ionicons name="close-circle" size={22} color="#FF4444" /></TouchableOpacity>
      </View>
    </View>
  );

  return (
    <View style={{ flex: 1 }}>
      <View style={styles.cartHeader}>
        <Text style={styles.cartTitle}>Корзина</Text>
        <TouchableOpacity onPress={() => { Alert.alert('Очистить корзину?', 'Все товары будут удалены', [{ text: 'Отмена', style: 'cancel' }, { text: 'Очистить', onPress: () => { onClear(); onShowToast('Корзина очищена', 'info'); } }]); }} style={styles.clearButton}>
          <Ionicons name="trash-outline" size={16} color="#FF6B00" /><Text style={styles.clearText}>Очистить</Text>
        </TouchableOpacity>
      </View>
      <FlatList data={cart} renderItem={renderItem} keyExtractor={(item) => item.id.toString()} contentContainerStyle={styles.cartList} showsVerticalScrollIndicator={false} />
      <View style={styles.totalBlock}>
        <View style={styles.totalRow}><Text style={styles.totalLabel}>Позиций:</Text><Text style={styles.totalValue}>{totalItems} шт.</Text></View>
        <View style={styles.totalRow}><Text style={styles.totalLabel}>Итого:</Text><Text style={styles.totalSum}>{total.toLocaleString('ru-RU')} ₽</Text></View>
        <TouchableOpacity style={styles.checkoutButton} onPress={onCheckout}><Text style={styles.checkoutButtonText}>Оформить заказ</Text></TouchableOpacity>
      </View>
    </View>
  );
}

  return (
    <TouchableOpacity style={styles.tabButton} onPress={onPress} activeOpacity={0.7}>
      <View style={styles.tabIconWrapper}>
        <Ionicons name={iconName} size={26} color={isActive ? '#FF6B00' : '#FFFFFF'} />

export default function App() {
  const [activeTab, setActiveTab] = useState('home');
  const [selectedCategory, setSelectedCategory] = useState(null);
  const [cart, setCart] = useState(null);
  const [orders, setOrders] = useState(null);
  const [cars, setCars] = useState(null);
  const [userName, setUserName] = useState(null);
  const [userPhone, setUserPhone] = useState(null);

  const [searchQuery, setSearchQuery] = useState('');
  const [highlightProductId, setHighlightProductId] = useState(null);
  const [selectedProduct, setSelectedProduct] = useState(null);
  const [selectedOrder, setSelectedOrder] = useState(null);

  const [showAddCarModal, setShowAddCarModal] = useState(false);
  const [newCarName, setNewCarName] = useState('');
  const [showEditCarModal, setShowEditCarModal] = useState(false);
  const [editingCar, setEditingCar] = useState(null);
  const [editCarName, setEditCarName] = useState('');
  const [editCarYear, setEditCarYear] = useState('');
  const [editCarVin, setEditCarVin] = useState('');
  
  const [showRequestModal, setShowRequestModal] = useState(false);

  const [toast, setToast] = useState({ visible: false, message: '', type: 'info' });
  const showToast = (message, type = 'info') => setToast({ visible: true, message, type });

  useEffect(() => {
    const loadData = async () => {
      const savedCart = await storage.loadCart();
      const savedOrders = await storage.loadOrders();
      const savedCars = await storage.loadCars();
      const savedProfile = await storage.loadProfile();
      setCart(savedCart !== null ? savedCart : []);
      setOrders(savedOrders !== null ? savedOrders : []);
      setCars(savedCars !== null ? savedCars : [{ id: 1, name: 'Tank 300', year: '2023 • 2.0T • 4WD', vin: 'LGWFF6A59PJ******' }]);
      setUserName(savedProfile?.name || '');
      setUserPhone(savedProfile?.phone || '');
    };
    loadData();
  }, []);

  useEffect(() => { if (cart !== null) storage.saveCart(cart); }, [cart]);
  useEffect(() => { if (orders !== null) storage.saveOrders(orders); }, [orders]);
  useEffect(() => { if (cars !== null) storage.saveCars(cars); }, [cars]);
  useEffect(() => { if (userName !== null && userPhone !== null) storage.saveProfile({ name: userName, phone: userPhone }); }, [userName, userPhone]);

  const handleCategoryPress = (categoryName, productId = null) => { setSelectedCategory(categoryName); setHighlightProductId(productId); setActiveTab('catalog'); setSearchQuery(''); };
  const handleBackToHome = () => { setSelectedCategory(null); setHighlightProductId(null); setActiveTab('home'); };
  const handleOpenAddCar = () => { setNewCarName(''); setShowAddCarModal(true); };
  const handleCloseAddCar = () => { setShowAddCarModal(false); setNewCarName(''); };
  const handleConfirmAddCar = () => {
    if (newCarName.trim().length < 2) { showToast('Введите марку и модель', 'error'); return; }
    setCars((prev) => [...prev, { id: Date.now(), name: newCarName.trim(), year: 'Год не указан', vin: '' }]);
    setShowAddCarModal(false);
    showToast('Автомобиль добавлен в гараж', 'success');
  };
  const handleOpenEditCar = (car) => { setEditingCar(car); setEditCarName(car.name); setEditCarYear(car.year); setEditCarVin(car.vin || ''); setShowEditCarModal(true); };
  const handleCloseEditCar = () => { setShowEditCarModal(false); setEditingCar(null); };
  const handleConfirmEditCar = () => {
    if (editCarName.trim().length < 2) { showToast('Введите марку и модель', 'error'); return; }
    setCars((prev) => prev.map((c) => c.id === editingCar.id ? { ...editingCar, name: editCarName.trim(), year: editCarYear.trim() || 'Год не указан', vin: editCarVin.trim() } : c));
    setShowEditCarModal(false);
    showToast('Данные автомобиля обновлены', 'success');
  };
  const handleRemoveCar = (carId) => setCars((prev) => prev.filter((c) => c.id !== carId));
  const handleAddToCart = (product) => setCart((prev) => {
    const existing = prev.find((item) => item.id === product.id);
    return existing ? prev.map((item) => item.id === product.id ? { ...item, quantity: item.quantity + 1 } : item) : [...prev, { ...product, quantity: 1 }];
  });
  const handleRemoveFromCart = (productId) => setCart((prev) => {
    const item = prev.find((i) => i.id === productId);
    return item && item.quantity > 1 ? prev.map((i) => i.id === productId ? { ...i, quantity: i.quantity - 1 } : i) : prev.filter((i) => i.id !== productId);
  });
  const handleRemoveItem = (productId) => setCart((prev) => prev.filter((item) => item.id !== productId));
  const handleClearCart = () => setCart([]);
  const handleCheckout = () => { if (cart.length === 0) { showToast('Добавьте товары в корзину', 'error'); return; } setActiveTab('checkout'); };
  
  const handleSubmitOrder = async (orderData) => {
    const orderId = genOrderId(orders);
    const itemsText = orderData.items.map(i => `${i.name} (${i.article}) x${i.quantity}`).join('; ');
    
    const newOrder = { 
      id: orderId, 
      date: new Date().toLocaleDateString('ru-RU'), 
      itemsCount: orderData.totalItems, 
      total: orderData.total, 
      status: 'В обработке', 
      customer: orderData.name, 
      phone: orderData.phone, 
      address: orderData.address, 
      transportCompany: orderData.transportCompany, 
      items: orderData.items || [] 
    };
    
    setOrders((prev) => [newOrder, ...prev]);
    setCart([]);
    setActiveTab('success');

    try {
      await fetch(SCRIPT_URL, {
        method: 'POST',
        mode: 'no-cors',
        headers: { 'Content-Type': 'text/plain;charset=utf-8' },
        body: JSON.stringify({
          type: 'order',
          orderId: String(orderId),
          name: orderData.name,
          phone: orderData.phone,
          address: orderData.address,
          tc: orderData.transportCompany,
          items: itemsText,
          total: orderData.total
        }),
      });
      console.log('✅ Заказ отправлен в таблицу');
    } catch (error) {
      console.error('❌ Ошибка отправки заказа:', error);
    }
  };

  // ОТМЕНА: меняет статус в таблице на "Отменён", но оставляет в истории приложения
  const handleCancelOrder = async (orderId) => {
    setOrders((prev) => prev.map((order) => order.id === orderId ? { ...order, status: 'Отменен' } : order));
    
    try {
      await fetch(SCRIPT_URL, {
        method: 'POST',
        mode: 'no-cors',
        headers: { 'Content-Type': 'text/plain;charset=utf-8' },
        body: JSON.stringify({
          type: 'cancel_order',
          orderId: String(orderId)
        }),
      });
      console.log('✅ Сигнал отмены отправлен');
    } catch (error) {
      console.error('❌ Ошибка отмены:', error);
    }
  };

  // УДАЛЕНИЕ: убирает из истории приложения И помечает как "Отменён" в таблице
  const handleDeleteOrder = async (orderId) => {
    setOrders((prev) => prev.filter((order) => order.id !== orderId));
    
    try {
      await fetch(SCRIPT_URL, {
        method: 'POST',
        mode: 'no-cors',
        headers: { 'Content-Type': 'text/plain;charset=utf-8' },
        body: JSON.stringify({
          type: 'delete_order',
          orderId: String(orderId)
        }),
      });
      console.log('✅ Сигнал удаления отправлен');
    } catch (error) {
      console.error('❌ Ошибка удаления:', error);
    }
  };

  const handleGoHome = () => setActiveTab('home');
  
  const handleRequestPart = async (requestData) => {
    showToast('Отправка запроса...', 'info');
    
    try {
      await fetch(SCRIPT_URL, {
        method: 'POST',
        mode: 'no-cors',
        headers: { 'Content-Type': 'text/plain;charset=utf-8' },
        body: JSON.stringify({
          type: 'request',
          car: requestData.car,
          part: requestData.part,
          comment: requestData.comment,
          phone: requestData.phone
        }),
      });
      showToast('Запрос успешно отправлен! Мы свяжемся с вами.', 'success');
    } catch (error) {
      console.error('❌ Ошибка:', error);
      showToast('Ошибка сети. Проверьте интернет.', 'error');
    }
  };

  const totalCartItems = useMemo(() => cart ? cart.reduce((sum, item) => sum + item.quantity, 0) : 0, [cart]);

  if (cart === null || cars === null) {
    return (<View style={styles.loadingContainer}><Text style={styles.loadingText}>Загрузка данных...</Text></View>);
  }

  const renderScreen = () => {
    switch (activeTab) {
      case 'home': return <HomeScreen onCategoryPress={handleCategoryPress} onViewProduct={setSelectedProduct} searchQuery={searchQuery} onSearchChange={setSearchQuery} garageCars={cars} />;
      case 'catalog': return <CatalogScreen categoryName={selectedCategory} onBack={handleBackToHome} cart={cart} onAddToCart={handleAddToCart} onRemoveFromCart={handleRemoveFromCart} highlightProductId={highlightProductId} garageCars={cars} onViewProduct={setSelectedProduct} onShowToast={showToast} />;
      case 'cart': return <CartScreen cart={cart} onRemove={handleRemoveItem} onClear={handleClearCart} onCheckout={handleCheckout} onShowToast={showToast} />;
      case 'checkout': return <CheckoutScreen cart={cart} onBack={() => setActiveTab('cart')} onSubmitOrder={handleSubmitOrder} defaultName={userName} defaultPhone={userPhone} onShowToast={showToast} />;
      case 'success': return <OrderSuccessScreen onGoHome={handleGoHome} />;
      case 'profile': return <ProfileScreen orders={orders} onCancelOrder={handleCancelOrder} onDeleteOrder={handleDeleteOrder} cars={cars} onAddCar={handleOpenAddCar} onRemoveCar={handleRemoveCar} onEditCar={handleOpenEditCar} userName={userName} setUserName={setUserName} userPhone={userPhone} setUserPhone={setUserPhone} onViewOrder={setSelectedOrder} onShowToast={showToast} onRequestPart={() => setShowRequestModal(true)} />;
      default: return <HomeScreen onCategoryPress={handleCategoryPress} onViewProduct={setSelectedProduct} searchQuery={searchQuery} onSearchChange={setSearchQuery} garageCars={cars} />;
    }
  };

  const showTabBar = !['checkout', 'success'].includes(activeTab);
  const firstCarName = cars.length > 0 ? cars[0].name : '';

  return (
    <ImageBackground source={require('./assets/tank.jpg')} style={styles.background} resizeMode="cover">
      <View style={styles.overlay} />
      <SafeAreaView style={styles.container}>
        <StatusBar barStyle="light-content" backgroundColor="transparent" translucent />
        <View style={styles.content}>{renderScreen()}</View>
        {showTabBar && (
          <View style={styles.tabBar}>
            <TabButton label="Главная" iconName="home" isActive={activeTab === 'home'} onPress={() => { setSelectedCategory(null); setHighlightProductId(null); setSearchQuery(''); setActiveTab('home'); }} />
            <TabButton label="Корзина" iconName="cart" isActive={activeTab === 'cart'} onPress={() => setActiveTab('cart')} badge={totalCartItems} />
            <TabButton label="Профиль" iconName="person" isActive={activeTab === 'profile'} onPress={() => setActiveTab('profile')} />
          </View>
        )}
      </SafeAreaView>

      <ToastNotification message={toast.message} type={toast.type} visible={toast.visible} onHide={() => setToast({ ...toast, visible: false })} />

      <RequestPartModal 
        visible={showRequestModal} 
        onClose={() => setShowRequestModal(false)} 
        onSubmit={handleRequestPart}
        defaultCar={firstCarName}
        defaultPhone={userPhone}
      />

      <ProductDetailModal product={selectedProduct} onClose={() => setSelectedProduct(null)} onAddToCart={handleAddToCart} onShowToast={showToast} />
      <OrderDetailModal order={selectedOrder} onClose={() => setSelectedOrder(null)} onCancelOrder={handleCancelOrder} onDeleteOrder={handleDeleteOrder} onShowToast={showToast} />

      <Modal visible={showAddCarModal} transparent animationType="fade" onRequestClose={handleCloseAddCar}>
        <View style={styles.modalOverlay}>
          <View style={styles.modalContent}>
            <Text style={styles.modalTitle}>Добавить автомобиль</Text>
            <TextInput style={styles.modalInput} placeholder="Например: Toyota Camry" value={newCarName} onChangeText={setNewCarName} autoFocus />
            <View style={styles.modalButtons}>
              <TouchableOpacity style={styles.modalButtonCancel} onPress={handleCloseAddCar}><Text style={styles.modalButtonCancelText}>Отмена</Text></TouchableOpacity>
              <TouchableOpacity style={styles.modalButtonConfirm} onPress={handleConfirmAddCar}><Text style={styles.modalButtonConfirmText}>Добавить</Text></TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>

      <Modal visible={showEditCarModal} transparent animationType="fade" onRequestClose={handleCloseEditCar}>
        <View style={styles.modalOverlay}>
          <View style={styles.modalContent}>
            <Text style={styles.modalTitle}>Редактировать автомобиль</Text>
            <Text style={styles.modalLabel}>Марка и модель *</Text>
            <TextInput style={styles.modalInput} value={editCarName} onChangeText={setEditCarName} autoFocus />
            <Text style={styles.modalLabel}>Год и характеристики</Text>
            <TextInput style={styles.modalInput} value={editCarYear} onChangeText={setEditCarYear} />
            <Text style={styles.modalLabel}>VIN номер</Text>
            <TextInput style={styles.modalInput} value={editCarVin} onChangeText={setEditCarVin} />
            <View style={styles.modalButtons}>
              <TouchableOpacity style={styles.modalButtonCancel} onPress={handleCloseEditCar}><Text style={styles.modalButtonCancelText}>Отмена</Text></TouchableOpacity>
              <TouchableOpacity style={styles.modalButtonConfirm} onPress={handleConfirmEditCar}><Text style={styles.modalButtonConfirmText}>Сохранить</Text></TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>
    </ImageBackground>
  );
}
